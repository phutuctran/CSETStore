            if(activeNav) activeNav.className = "nav-btn px-5 py-2.5 rounded-xl text-sm font-semibold text-white bg-blue-900 shadow-sm cursor-pointer transition";

            if (tabId === 'home') renderHomeProducts();
            if (tabId === 'shop') renderShopProducts(products);
            if (tabId === 'admin') renderAdminTable();
        }

        function formatPrice(num) {
            return num.toLocaleString('vi-VN') + ' VNĐ';
        }

        function renderHomeProducts() {
            const grid = document.getElementById('home-products-grid');
            if(!grid) return;
            grid.innerHTML = products.slice(0, 4).map(p => `
                <div onclick="openProductModal(${p.id})" class="glass-panel rounded-2xl overflow-hidden border border-gray-200 flex flex-col justify-between bg-white shadow-sm hover:shadow-md transition cursor-pointer">
                    <img src="${p.image}" class="h-48 w-full object-cover">
                    <div class="p-5 space-y-3">
                        <h3 class="font-bold text-blue-950 line-clamp-1">${p.name}</h3>
                        <p class="text-blue-900 font-extrabold">${formatPrice(p.price)}</p>
                        <button type="button" class="w-full py-2.5 rounded-xl bg-gray-100 hover:bg-blue-900 hover:text-white font-semibold text-xs text-blue-950 transition">Xem Chi Tiết</button>
                    </div>
                </div>
            `).join('');
        }

        function renderShopProducts(items) {
            const grid = document.getElementById('shop-products-grid');
            if(!grid) return;
            grid.innerHTML = items.map(p => `
                <div onclick="openProductModal(${p.id})" class="glass-panel rounded-2xl overflow-hidden border border-gray-200 flex flex-col justify-between bg-white shadow-sm hover:shadow-md transition cursor-pointer">
                    <img src="${p.image}" class="h-48 w-full object-cover">
                    <div class="p-5 space-y-3">
                        <h3 class="font-bold text-blue-950 line-clamp-1">${p.name}</h3>
                        <p class="text-blue-900 font-extrabold">${formatPrice(p.price)}</p>
                        <button type="button" class="w-full py-2.5 rounded-xl bg-blue-900 text-white font-bold text-xs transition">Xem Chi Tiết</button>
                    </div>
                </div>
            `).join('');
        }

        function applyFilters() {
            const cat = document.getElementById('filter-category').value;
            const search = document.getElementById('filter-search').value.toLowerCase();
            let filtered = products.filter(p => {
                const matchCat = (cat === 'all' || p.category === cat);
                const matchSearch = p.name.toLowerCase().includes(search);
                return matchCat && matchSearch;
            });
            renderShopProducts(filtered);
        }

        function openProductModal(id) {
            currentModalProduct = products.find(p => p.id === id);
            if(!currentModalProduct) return;
            
            document.getElementById('modal-title').innerText = currentModalProduct.name;
            document.getElementById('modal-price').innerText = formatPrice(currentModalProduct.price);
            document.getElementById('modal-desc').innerText = currentModalProduct.description || "Sản phẩm in 3D chất lượng cao tại CSET Store.";
            document.getElementById('modal-category').innerText = currentModalProduct.category.toUpperCase();
            document.getElementById('modal-img').src = currentModalProduct.image;
            
            document.getElementById('product-modal').classList.remove('hidden');
        }

        function closeProductModal() { 
            document.getElementById('product-modal').classList.add('hidden'); 
        }

        function addModalProductToCart() {
            if(!currentModalProduct) return;
            cart.push(currentModalProduct);
            updateCartBadge();
            closeProductModal();
            alert('Đã thêm sản phẩm vào giỏ hàng thành công!');
        }

        function updateCartBadge() { 
            const badge = document.getElementById('cart-badge');
            if(badge) badge.innerText = cart.length; 
        }

        function openCartModal() {
            const list = document.getElementById('cart-items-list');
            if(!list) return;
            let total = 0;
            list.innerHTML = cart.map((item, index) => {
                total += item.price;
                return `
                    <div class="flex items-center justify-between bg-gray-50 p-3 rounded-xl border border-gray-200">
                        <div class="flex items-center space-x-3">
                            <img src="${item.image}" class="w-12 h-12 rounded-lg object-cover">
                            <div>
                                <p class="text-blue-950 text-sm font-bold">${item.name}</p>
                                <p class="text-blue-900 text-xs font-semibold">${formatPrice(item.price)}</p>
                            </div>
                        </div>
                        <button type="button" onclick="removeFromCart(${index})" class="text-red-500 text-xs hover:text-red-700 cursor-pointer"><i class="fa-solid fa-trash"></i> Xóa</button>
                    </div>
                `;
            }).join('');
            document.getElementById('cart-total-price').innerText = formatPrice(total);
            document.getElementById('cart-modal').classList.remove('hidden');
        }

        function closeCartModal() { document.getElementById('cart-modal').classList.add('hidden'); }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCartBadge();
            openCartModal();
        }

        function processCheckout() {
            if(cart.length === 0) return alert('Giỏ hàng trống!');
            
            const name = document.getElementById('checkout-name').value.trim();
            const phone = document.getElementById('checkout-phone').value.trim();
            const address = document.getElementById('checkout-address').value.trim();

            if(!name || !phone || !address) {
                return alert('Vui lòng điền đầy đủ thông tin giao hàng (Họ tên, Số điện thoại, Địa chỉ)!');
            }

            alert(`Cảm ơn bạn ${name}! Đặt hàng thành công.\nCSET Store sẽ liên hệ qua số điện thoại ${phone} và giao hàng đến địa chỉ: ${address} trong thời gian sớm nhất.`);
            cart = [];
            updateCartBadge();
            closeCartModal();
        }

        // Xử lý file in 3D & tính toán chi phí
        function handleFileSelected(e) {
            const file = e.target.files[0];
            if(!file) return;
            uploadedFileName = file.name;
            const displayEl = document.getElementById('file-name-display');
            if(displayEl) displayEl.innerText = `Đã nhận file: ${file.name}`;
            
            let sizeKB = file.size / 1024;
            baseCalculatedGrams = Math.max(5, Math.min(600, Math.round(sizeKB * 0.15)));

            recalculateStats();
            alert(`Đã phân tích file "${file.name}" thành công! Hệ thống đã tính toán lượng nhựa và thời gian in.`);
        }

        function recalculateStats() {
            const infillEl = document.getElementById('infill-select');
            const infillPercent = infillEl ? parseInt(infillEl.value) / 100 : 0.2;
            const finalGrams = Math.round(baseCalculatedGrams * (0.4 + 0.6 * infillPercent));
            
            const totalMinutes = finalGrams * 8;
            const hours = (totalMinutes / 60).toFixed(1);
            const totalPrice = finalGrams * 1500;

            const gEl = document.getElementById('stat-grams');
            const hEl = document.getElementById('stat-hours');
            const pEl = document.getElementById('custom-price-display');

            if(gEl) gEl.innerText = `${finalGrams} gam`;
            if(hEl) hEl.innerText = `${hours} giờ`;
            if(pEl) pEl.innerText = formatPrice(totalPrice);
        }

        function addCustomToCart() {
            const gEl = document.getElementById('stat-grams');
            const pEl = document.getElementById('custom-price-display');
            if(!gEl || !pEl) return;

            const gramsText = gEl.innerText;
            const priceText = pEl.innerText;
            const price = parseInt(priceText.replace(/\D/g, ''));
            const fileName = uploadedFileName || 'File_In_3D_Custom';

            cart.push({
                name: `In 3D Theo Yêu Cầu (${fileName} - ${gramsText})`,
                price: price,
                image: "https://images.unsplash.com/photo-1581092160607-ee22621dd758?w=600&auto=format&fit=crop&q=80"
            });
            updateCartBadge();
            alert('Đã thêm dịch vụ in 3D vào giỏ hàng CSET Store!');
        }

        function handleImageUpload(e) {
            const file = e.target.files[0];
            if(file) {
                const reader = new FileReader();
                reader.onload = function(evt) { uploadedImageBase64 = evt.target.result; }
                reader.readAsDataURL(file);
            }
        }

        function handleAddNewProduct(e) {
            e.preventDefault();
            const name = document.getElementById('admin-name').value;
            const price = parseInt(document.getElementById('admin-price').value);
            const category = document.getElementById('admin-category').value;
            const finalImage = uploadedImageBase64 || "https://images.unsplash.com/photo-1607604276583-eef5d076aa5f?w=600&auto=format&fit=crop&q=80";

            products.push({ id: Date.now(), name, category, price, image: finalImage, description: "Sản phẩm in 3D chất lượng cao được thiết kế tại CSET Store." });
            saveProductsToStorage();
            renderAdminTable();
            renderHomeProducts();
            alert('Thêm sản phẩm mới thành công!');
            e.target.reset();
            uploadedImageBase64 = '';
        }

        function deleteProduct(id) {
            if(confirm('Bạn muốn xóa sản phẩm này khỏi CSET Store?')) {
                products = products.filter(p => p.id !== id);
                saveProductsToStorage();
                renderAdminTable();
                renderHomeProducts();
            }
        }

        function renderAdminTable() {
            const tbody = document.getElementById('admin-product-table');
            if(!tbody) return;
            tbody.innerHTML = products.map(p => `
                <tr class="border-b border-gray-200">
                    <td class="p-3"><img src="${p.image}" class="w-10 h-10 rounded object-cover"></td>
                    <td class="p-3 font-bold text-blue-950">${p.name}</td>
                    <td class="p-3 text-blue-900 font-semibold">${formatPrice(p.price)}</td>
                    <td class="p-3"><button type="button" onclick="deleteProduct(${p.id})" class="text-red-500 hover:text-red-700 font-semibold cursor-pointer"><i class="fa-solid fa-trash"></i> Xóa</button></td>
                </tr>
            `).join('');
        }

        document.addEventListener('DOMContentLoaded', function() {
            renderHomeProducts();
            recalculateStats();
        });
